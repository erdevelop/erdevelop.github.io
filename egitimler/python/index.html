gun2.py:

# gun2.py - panel listesi özeti
from collections import Counter

paneller = [
    {"no": "P-001", "g": 1500, "y": 3200},
    {"no": "P-002", "g": 1500, "y": 3200},
    {"no": "P-003", "g": 1200, "y": 3200},
    {"no": "P-004", "g": 900,  "y": 3200},
    {"no": "P-005", "g": 1500, "y": 2800},
]


def panel_tip(genislik):
    if genislik > 1400:
        return "G01"
    elif genislik > 900:
        return "G02"
    return "G03"


for p in paneller:
    p["tip"] = panel_tip(p["g"])
    p["alan"] = p["g"] * p["y"] / 1e6

sayim = Counter(p["tip"] for p in paneller)
alanlar = {}
for p in paneller:
    alanlar[p["tip"]] = alanlar.get(p["tip"], 0) + p["alan"]

print("Tip   Adet   Alan (m²)")
for tip in sorted(sayim):
    print(f"{tip:<5} {sayim[tip]:>4} {alanlar[tip]:>11.2f}")
print(f"Toplam {len(paneller)} panel, {sum(alanlar.values()):.2f} m²")

en_buyuk = max(paneller, key=lambda p: p["alan"])
print("En büyük panel:", en_buyuk["no"])
  1. Counter, öğeleri sayar: Counter({'G01': 3, 'G02': 1, 'G03': 1}).
  2. alanlar.get(tip, 0) + ... kalıbı: anahtar ilk kez görülüyorsa 0'dan başlar. Gruplayarak toplamanın temel yoludur.
  3. key=lambda p: p["alan"]: max ve sorted'a neye göre karşılaştıracağını söyler. Aynı kalıp sıralamada da kullanılır: sorted(paneller, key=lambda p: (p["y"], p["g"])).
  4. {tip:<5} sola, {adet:>4} sağa yaslar; konsolda düzgün tablo çıkar.

import csv
from pathlib import Path
from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill

KLASOR = Path(__file__).parent

with open(KLASOR / "paneller.csv", encoding="utf-8-sig", newline="") as f:
    paneller = list(csv.DictReader(f, delimiter=";"))

wb = Workbook()
ws = wb.active
ws.title = "Paneller"
ws.append(["Panel No", "Genişlik", "Yükseklik", "Cam", "Alan (m²)"])

toplam = 0
for p in paneller:
    g, y = int(p["GENISLIK"]), int(p["YUKSEKLIK"])
    alan = round(g * y / 1e6, 3)
    toplam += alan
    ws.append([p["PANEL_NO"], g, y, p["CAM_KODU"], alan])

ws.append([])
ws.append(["TOPLAM", None, None, None, round(toplam, 3)])

# biçim
for hucre in ws[1]:

wb.save(KLASOR / "panel_raporu.xlsx")

# gun4.py - metraj / sipariş karşılaştırması
import pandas as pd
from pathlib import Path

KLASOR = Path(__file__).parent
ANAHTAR = ["CAM_KODU", "GENISLIK", "YUKSEKLIK"]

proje = pd.read_csv(KLASOR / "paneller.csv", sep=";", encoding="utf-8-sig")
siparis = pd.read_csv(KLASOR / "siparis.csv", sep=";", encoding="utf-8-sig")

metraj = proje.groupby(ANAHTAR).size().reset_index(name="PROJE")

k = metraj.merge(siparis, on=ANAHTAR, how="outer").fillna(0)
k[["PROJE", "SIPARIS"]] = k[["PROJE", "SIPARIS"]].astype(int)
k["FARK"] = k["SIPARIS"] - k["PROJE"]
k["DURUM"] = "TAMAM"
k.loc[k["FARK"] < 0, "DURUM"] = "EKSİK SİPARİŞ"
k.loc[k["FARK"] > 0, "DURUM"] = "FAZLA SİPARİŞ"

sorunlu = k[k["FARK"] != 0]
print(sorunlu.to_string(index=False))

with pd.ExcelWriter(KLASOR / "karsilastirma.xlsx") as yazici:
    k.to_excel(yazici, sheet_name="Tümü", index=False)
    sorunlu.to_excel(yazici, sheet_name="Farklar", index=False)
    (proje.groupby("CAM_KODU").size().reset_index(name="ADET")
          .to_excel(yazici, sheet_name="Cam Özeti", index=False))
print("karsilastirma.xlsx yazıldı")
  1. size() her gruptaki satır sayısını verir; reset_index(name=...) sonucu yeniden tabloya çevirir.
  2. merge(..., how="outer"): Excel'deki DÜŞEYARA'nın iki yönlü hali. Sadece projede ya da sadece siparişte olan satırlar da gelir; eşi olmayan taraf boş (NaN) kalır, fillna(0) ile sıfır yapılır.
  3. k.loc[koşul, "DURUM"] = ...: koşulu sağlayan satırlarda sütuna değer yazar.
  4. Örnek veride C1 1500x3200 bir eksik, C3 bir fazla, C2 900x3200 hiç sipariş edilmemiş, C4 projede yok. Program dördünü de bulur.

Polyline'ın köşeleri: e.get_points("xy") → [(x, y), ...]. Alanı ayakkabı bağı formülüyle (shoelace) hesaplarız; bu formül her basit kapalı çokgende çalışır:

def alan(noktalar):
    """Kapalı çokgen alanı (birim²)."""
    s = 0.0
    for (x1, y1), (x2, y2) in zip(noktalar, noktalar[1:] + noktalar[:1]):
        s += x1 * y2 - x2 * y1
    return abs(s) / 2


def merkez(noktalar):
    """Köşelerin ortalaması (dikdörtgen panelde gerçek merkez)."""
    xs, ys = zip(*noktalar)
    return sum(xs) / len(xs), sum(ys) / len(ys)

zip(noktalar, noktalar[1:] + noktalar[:1]) her köşeyi bir sonrakiyle eşler; son köşe ilk köşeyle kapanır.

Not: Yay (bulge) segmentli polyline'larda bu formül yayları düz kabul eder. Yaylı paneller için ezdxf'in ezdxf.path modülüyle eğriyi önce noktalara böl.

panel_lib.py:

"""Panel hesapları - Cephe Python araçları."""
import ezdxf
from ezdxf.enums import TextEntityAlignment

KAT_TOLERANS = 100


def alan(noktalar):
    s = 0.0
    for (x1, y1), (x2, y2) in zip(noktalar, noktalar[1:] + noktalar[:1]):
        s += x1 * y2 - x2 * y1
    return abs(s) / 2


def merkez(noktalar):
    xs, ys = zip(*noktalar)
    return sum(xs) / len(xs), sum(ys) / len(ys)


def panelleri_oku(dxf_yolu, katman="C-CAM"):
    """DXF'teki kapalı polyline'ları sıralı panel listesi olarak döndürür."""
    doc = ezdxf.readfile(dxf_yolu)
    paneller = []
    for e in doc.modelspace().query(f'LWPOLYLINE[layer=="{katman}"]'):
        if not e.closed:
            continue
        pts = [tuple(p) for p in e.get_points("xy")]
        xs, ys = zip(*pts)
        paneller.append({"merkez": merkez(pts), "alan": alan(pts) / 1e6,
                         "g": max(xs) - min(xs), "y": max(ys) - min(ys)})
    paneller.sort(key=lambda p: (round(p["merkez"][1] / KAT_TOLERANS), p["merkez"][0]))
    for i, p in enumerate(paneller, start=1):
        p["no"] = f"P-{i:03d}"
    return doc, paneller


def numara_yaz(doc, paneller, yukseklik=125, katman="C-YAZI"):
    if katman not in doc.layers:
        doc.layers.add(katman, color=2)
    msp = doc.modelspace()
    for p in paneller:
        msp.add_text(p["no"], height=yukseklik, dxfattribs={"layer": katman}) \
           .set_placement(p["merkez"], align=TextEntityAlignment.MIDDLE_CENTER)


if __name__ == "__main__":
    from pathlib import Path
    _, ps = panelleri_oku(Path(__file__).parent / "cephe.dxf")
    print(len(ps), "panel okundu (modül testi)")
  1. Başka bir dosyadan kullanmak: from panel_lib import panelleri_oku, numara_yaz.
  2. if __name__ == "__main__": bloğu sadece dosya doğrudan çalıştırıldığında çalışır, import edildiğinde çalışmaz. Modülü test etmenin kolay yolu.

rhino_panel.py:

# rhino_panel.py - Rhino 8 ScriptEditor (Python 3)
import csv
import rhinoscriptsyntax as rs

KAT_TOLERANS = 100

ids = rs.GetObjects("Cam panellerini seç", rs.filter.surface | rs.filter.polysurface,
                    preselect=True)
if ids:
    paneller = []
    for i in ids:
        a = rs.SurfaceArea(i)[0]
        m = rs.SurfaceAreaCentroid(i)[0]
        paneller.append({"id": i, "alan": a / 1e6, "x": m.X, "z": m.Z, "m": m})

    # cephe XZ düzleminde: önce kat (z), sonra x
    paneller.sort(key=lambda p: (round(p["z"] / KAT_TOLERANS), p["x"]))

    rs.EnableRedraw(False)
    for n, p in enumerate(paneller, start=1):
        no = "P-{:03d}".format(n)
        p["no"] = no
        rs.ObjectName(p["id"], no)
        rs.AddTextDot(no, p["m"])
    rs.EnableRedraw(True)

    yol = rs.SaveFileName("Panel listesi", "CSV (*.csv)|*.csv||", filename="rhino_paneller")
    if yol:
        with open(yol, "w", encoding="utf-8-sig", newline="") as f:
            w = csv.writer(f, delimiter=";")
            w.writerow(["PANEL_NO", "ALAN_M2", "X", "Z"])
            for p in paneller:
                w.writerow([p["no"], round(p["alan"], 3), round(p["x"]), round(p["z"])])
    print("{} panel numaralandı".format(len(paneller)))
  1. Cephe XZ düzleminde varsayıldı (ön görünüş). Cephen XY düzlemindeyse m.Z yerine m.Y kullan.
  2. rs.ObjectName, numarayı nesnenin adına da yazar: Rhino'da SelName P-012 ile o paneli bulursun ve isim Revit'e ya da IFC'ye taşınır.
  3. AddTextDot, her ölçekte okunur kalan nokta etiketidir.
  4. EnableRedraw(False): yüzlerce nesnede ekranın her adımda yeniden çizilmesini önler; işlem çok hızlanır. Sonunda True yapmayı unutma.
  5. CSV, 4. gündeki pandas karşılaştırmasına doğrudan girebilir.

Not: Rhino 8'de kendi Python 3 ortamına paket de kurulabilir: betiğin başına # r: pandas yazınca ScriptEditor paketi kurar.

2. Grasshopper'da Python bileşeni (1,5 saat)

# Grasshopper Python 3 bileşeni
# girdi: srf (list of Surface/Brep), tol (float)  çıktı: no, alan, nokta
import Rhino.Geometry as rg

tol = tol or 100
kayit = []
for s in srf:
    b = s.ToBrep() if hasattr(s, "ToBrep") else s
    amp = rg.AreaMassProperties.Compute(b)
    c = amp.Centroid
    kayit.append((round(c.Z / tol), c.X, amp.Area / 1e6, c))

kayit.sort(key=lambda k: (k[0], k[1]))
no = ["P-{:03d}".format(i) for i in range(1, len(kayit) + 1)]
alan = [round(k[2], 3) for k in kayit]
nokta = [k[3] for k in kayit]
  1. no ve nokta çıktılarını bir Text Tag 3D bileşenine bağla: numaralar ekranda görünür. alan'ı Panel'e bağla.
