3. Veri sınıfı (1 saat)

from dataclasses import dataclass

YOGUNLUK = 2.5  # kg/m², her 1 mm cam kalınlığı için

@dataclass
class Cam:
    poz: str
    en: float          # mm
    boy: float         # mm
    katman: str        # "8+6" gibi
    adet: int = 1
    id: int | None = None

    @property
    def kalinlik(self) -> float:
        return sum(float(k.replace(",", ".")) for k in self.katman.split("+"))

    @property
    def alan(self) -> float:          # m², tek cam
        return self.en * self.boy / 1e6

    @property
    def agirlik(self) -> float:       # kg, tek cam
        return self.alan * self.kalinlik * YOGUNLUK
  1. @dataclass, alanları yazmayı ve yazdırmayı otomatik yapar.
  2. @property ile hesaplanan değerler bir alan gibi okunur: cam.agirlik. Hiçbir yerde saklanmaz, her zaman güncel.
  3. Tip ipuçları (float, int | None) VS Code'un hataları önceden göstermesini sağlar.

Doğrulama da hesap katmanındadır; arayüze değil, mesaj listesine döner:

def dogrula(cam: Cam) -> list[str]:
    hatalar = []
    if not cam.poz.strip():
        hatalar.append("Poz boş olamaz.")
    if cam.en <= 0 or cam.boy <= 0:
        hatalar.append("En ve boy pozitif olmalı.")
    try:
        if cam.kalinlik <= 0:
            hatalar.append("Katman kalınlıkları pozitif olmalı.")
    except ValueError:
        hatalar.append("Katmanları 8+6 gibi yaz.")
    if cam.adet < 1:
        hatalar.append("Adet en az 1 olmalı.")
    return hatalar

2. Depo sınıfı (1,5 saat)

Arayüz SQL bilmez; yalnızca ekle, guncelle, sil, hepsi metotlarını çağırır:

import sqlite3
from hesap import Cam

SEMA = """
CREATE TABLE IF NOT EXISTS camlar (
  id      INTEGER PRIMARY KEY,
  poz     TEXT NOT NULL,
  en      REAL NOT NULL CHECK (en > 0),
  boy     REAL NOT NULL CHECK (boy > 0),
  katman  TEXT NOT NULL,
  adet    INTEGER NOT NULL CHECK (adet >= 1)
);
"""

class Depo:
    def __init__(self, yol):
        self.con = sqlite3.connect(yol)
        self.con.executescript(SEMA)          # ilk açılışta tabloyu kurar

    def hepsi(self):
        rows = self.con.execute("SELECT poz, en, boy, katman, adet, id FROM camlar ORDER BY poz").fetchall()
        return [Cam(*r) for r in rows]

    def ekle(self, c):
        with self.con:                        # işlem: hata olursa geri alınır, olmazsa kaydedilir
            cur = self.con.execute(
                "INSERT INTO camlar (poz, en, boy, katman, adet) VALUES (?, ?, ?, ?, ?)",
                (c.poz, c.en, c.boy, c.katman, c.adet))
        return cur.lastrowid

    def sil(self, cam_id):
        with self.con:
            self.con.execute("DELETE FROM camlar WHERE id = ?", (cam_id,))
  1. ? parametreleri: Değerleri asla metin birleştirerek sorguya koyma.
  2. with self.con: bir işlemdir; blok başarıyla biterse kaydeder (commit), hata olursa geri alır.
  3. CHECK kısıtları son savunma hattıdır: arayüzdeki doğrulama atlanırsa veritabanı yine reddeder (CHECK constraint failed: en > 0).

1. DXF'ten cam ölçüleri (1,5 saat)

Cephe çizimlerinde camlar çoğu zaman bir katmanda (ör. CAM) kapalı polilinler olarak çizilir. Her kapalı poliline sınır kutusu (bounding box) uygula; eni ve boyu oradan al:

import ezdxf
from ezdxf import bbox
from hesap import Cam

def camlari_oku(dosya, katman="CAM", varsayilan_katman="8+6"):
    doc = ezdxf.readfile(dosya)
    msp = doc.modelspace()
    camlar = []
    for i, e in enumerate(msp.query(f'LWPOLYLINE[layer=="{katman}"]'), 1):
        if not e.closed:
            continue                          # açık çizgileri atla
        kutu = bbox.extents([e])
        en, boy = round(kutu.size.x), round(kutu.size.y)
        camlar.append(Cam(poz=f"D-{i:02d}", en=en, boy=boy, katman=varsayilan_katman))
    return camlar

Deneme: üç kapalı dikdörtgen (1500 × 2500, 1200 × 2000, 1500 × 900) ve bir açık çizgi içeren bir DXF'ten üç cam okunur; açık çizgi atlanır.

def dxf(camlar, dosya):

    x = 0
    for c in camlar:
        msp.add_lwpolyline([(x, 0), (x + c.en, 0), (x + c.en, c.boy), (x, c.boy)],
                           close=True, dxfattribs={"layer": "CAM"})
        msp.add_text(f"{c.poz}  {c.en:g}x{c.boy:g}  {c.katman}  x{c.adet}", height=50,
                     dxfattribs={"layer": "ETIKET"}).set_placement((x, -120))
        x += c.en + 300                       # aralarına 300 mm boşluk
    doc.saveas(dosya)
  1. units=4 dosyaya birimin mm olduğunu yazar; AutoCAD'de ölçü doğru gelir.
  2. Katman adları ve renkleri ekip standardına göre.
  3. Türkçe karakterli metinler için güncel DXF sürümleri (ezdxf'in varsayılanı) sorun çıkarmaz.

from openpyxl import Workbook

def excel(camlar, dosya):

3. PDF (1,5 saat)

from reportlab.lib import colors
from reportlab.lib.pagesizes import A4
from reportlab.lib.styles import getSampleStyleSheet
from reportlab.pdfbase import pdfmetrics
from reportlab.pdfbase.ttfonts import TTFont
from reportlab.platypus import SimpleDocTemplate, Paragraph, Table, TableStyle, Spacer

def pdf(camlar, dosya, font_yolu=None):
    yazi = "Helvetica"
    if font_yolu:                     # ör. C:/Windows/Fonts/arial.ttf
        pdfmetrics.registerFont(TTFont("Govde", font_yolu))
        yazi = "Govde"
    stil = getSampleStyleSheet()
    stil["Title"].fontName = yazi
    stil["Normal"].fontName = yazi
    veri = [BASLIK] + [satir(c) for c in camlar]
    tablo = Table(veri, repeatRows=1)              # başlık her sayfada tekrar
    tablo.setStyle(TableStyle([
        ("FONTNAME", (0, 0), (-1, -1), yazi),
        ("BACKGROUND", (0, 0), (-1, 0), colors.HexColor("#dde3ea")),
        ("GRID", (0, 0), (-1, -1), 0.5, colors.grey),
        ("ALIGN", (1, 1), (-1, -1), "RIGHT"),
    ]))
    SimpleDocTemplate(dosya, pagesize=A4).build(
        [Paragraph("Cam Listesi", stil["Title"]), Spacer(1, 12), tablo])

Türkçe karakter tuzağı: reportlab'in yerleşik yazı tipleri (Helvetica) ş, ğ, ı, İ harflerini doğru göstermez. Bir TTF yazı tipi kaydet (Windows'ta C:/Windows/Fonts/arial.ttf); paketlenmiş programda yazı tipini programla birlikte dağıtmak daha güvenlidir (7. gün).

1. Hatalar ve kayıt (1 saat)

Exe'de konsol yoktur; bir hata olduğunda program sessizce çöker ya da hiçbir şey yapmaz. Çözüm: kayıt dosyası ve genel hata yakalayıcı.

import logging, traceback
from tkinter import messagebox

logging.basicConfig(filename=ayar.KLASOR / "hata.log", level=logging.INFO,
                    format="%(asctime)s %(levelname)s %(message)s", encoding="utf-8")

def yakalanmayan(tur, deger, iz):
    logging.error("".join(traceback.format_exception(tur, deger, iz)))
    messagebox.showerror("Beklenmeyen hata", f"{deger}\n\nAyrıntılar: {ayar.KLASOR / 'hata.log'}")

uyg = Uygulama(db)
uyg.report_callback_exception = yakalanmayan    # buton ve menü olaylarındaki bütün hatalar

Kullanıcı bir hata bildirdiğinde hata.log dosyasını iste: ne olduğu, satır numarasıyla oradadır.
